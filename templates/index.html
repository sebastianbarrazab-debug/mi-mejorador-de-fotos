// ========================================================================= // MOTOR DE TEXTO INTELIGENTE (Corrección avanzada de modismos, preguntas y puntuación) // ========================================================================= function normalizarModismosTexto(texto) { let limpio = texto; const diccionario = [ { regex: /\bnc\b/gi, reemplazo: 'no sé' }, { regex: /\bq\b/gi, reemplazo: 'que' }, { regex: /\bxq\b/gi, reemplazo: 'porque' }, { regex: /\btb\b/gi, reemplazo: 'también' }, { regex: /\bx\b/gi, reemplazo: 'por' }, { regex: /\bpa\b/gi, reemplazo: 'para' }, { regex: /\bfavor\b/gi, reemplazo: 'por favor' } ]; diccionario.forEach(item => { limpio = limpio.replace(item.regex, item.reemplazo); }); return limpio; } // Corrector estricto de puntuación, espacios y delimitación de preguntas en español function corregirPuntuacionYEstructura(texto) { let limpio = texto.trim(); // 1. Corregir espacios antes de signos de puntuación limpio = limpio.replace(/\s+([,.;:?!])/g, '$1'); // 2. Asegurar espacio después de signos de puntuación si falta limpio = limpio.replace(/([,.;:?!])(?=[^\s])/g, '$1 '); // 3. Detección y corrección inteligente de preguntas (Añadir ¿ y ?) // Palabras interrogativas comunes en español const interrogativos = /\b(que|como|cuando|donde|quien|por que|cuanto|cual|cuales|porqué)\b/i; let frases = limpio.split(/([.?!]\s+|\n+)/); let resultadoFrases = []; for (let i = 0; i < frases.length; i++) { let frase = frases[i].trim(); if (!frase) continue; // Si la frase empieza con palabra interrogativa o tiene tono de pregunta pero le falta el ¿ o el ? let esInterrogativa = interrogativos.test(frase) || /^(es|son|tiene|tienes|puedes|sabes|quieres|haces|donde|cuando)\b/i.test(frase); if (esInterrogativa) { // Si no tiene signo de interrogación al inicio, se lo agregamos if (!frase.startsWith('¿') && !frase.startsWith('¡')) { // Buscar si hay una conjunción inicial o pronombre para ubicar bien el ¿ frase = '¿' + frase; } // Asegurar cierre con ? if (!frase.endsWith('?') && !frase.endsWith('!')) { frase = frase.replace(/[.,;:]*$/, '?'); } } else { // Si es frase normal, asegurar punto final si no tiene signo if (!/[.?!]$/.test(frase)) { frase += '.'; } } // Mayúscula inicial en cada oración frase = frase.charAt(0).toUpperCase() + frase.slice(1); resultadoFrases.push(frase); } return resultadoFrases.join(' '); } async function analizarTexto() { let textoCrudo = document.getElementById('input-texto').value.trim(); if (!textoCrudo) return alert("Por favor, ingresa un texto para analizar."); let texto = normalizarModismosTexto(textoCrudo); document.getElementById('stats-texto').style.display = 'flex'; document.getElementById('lbl-errores').innerText = "..."; document.getElementById('contenedor-errores').style.display = 'none'; document.getElementById('seccion-resultado-texto').style.display = 'none'; try { let response = await fetch('https://api.languagetool.org/v2/check', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ 'text': texto, 'language': 'es' }) }); let data = await response.json(); let matches = data.matches || []; document.getElementById('lbl-errores').innerText = matches.length; let contenedorLista = document.getElementById('lista-errores'); contenedorLista.innerHTML = ''; if (matches.length === 0) { contenedorLista.innerHTML = '
¡Tu texto está perfecto! 🎉 No se detectaron errores de gramática ni puntuación.
'; } else { matches.forEach((err, idx) => { let textoAfectado = texto.substring(err.offset, err.offset + err.length); let esGramatica = err.rule && err.rule.issueType === 'grammar'; let esPuntuacion = err.rule && err.rule.issueType === 'typographical'; let claseColor = esPuntuacion ? 'punctuation' : (esGramatica ? 'grammar' : ''); let tipoError = esPuntuacion ? 'Puntuación/Espaciado' : (esGramatica ? 'Gramática' : 'Ortografía/Modismo'); let colorTexto = esPuntuacion ? '#3b82f6' : (esGramatica ? '#f59e0b' : '#ef4444'); let sugerenciasHTML = err.replacements.slice(0, 4).map(rep => `` ).join(''); let card = document.createElement('div'); card.className = `error-card ${claseColor}`; card.innerHTML = `
Error #${idx + 1}: ${tipoError}
Encontrado en: ${textoAfectado || '(Espacio/Signo faltante)'}
${err.message}
${sugerenciasHTML ? `
Sugerencias:
${sugerenciasHTML}
` : ''} `; contenedorLista.appendChild(card); }); } document.getElementById('contenedor-errores').style.display = 'block'; } catch (e) { alert("Error al conectar con el servidor de análisis."); } } function aplicarCorreccion(offset, length, nuevoTexto) { let input = document.getElementById('input-texto'); let textoOriginal = input.value; input.value = textoOriginal.substring(0, offset) + nuevoTexto + textoOriginal.substring(offset + length); analizarTexto(); } async function mejorarTextoIA() { let textoCrudo = document.getElementById('input-texto').value.trim(); let estilo = document.getElementById('sel-estilo-ia').value; if (!textoCrudo) return alert("Por favor, ingresa un texto."); let texto = normalizarModismosTexto(textoCrudo); let output = document.getElementById('output-texto'); document.getElementById('seccion-resultado-texto').style.display = 'block'; output.value = "Procesando correcciones con contexto y puntuación..."; if (estilo === 'auto') { document.getElementById('lbl-titulo-resultado').innerText = "✨ Arreglado Automáticamente (Modismos, preguntas y puntuación):"; try { let response = await fetch('https://api.languagetool.org/v2/check', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ 'text': texto, 'language': 'es' }) }); let data = await response.json(); let matches = data.matches || []; let textoCorregido = texto; for (let i = matches.length - 1; i >= 0; i--) { let match = matches[i]; if (match.replacements && match.replacements.length > 0) { let sugerencia = match.replacements[0].value; textoCorregido = textoCorregido.substring(0, match.offset) + sugerencia + textoCorregido.substring(match.offset + match.length); } } // Aplicar el corrector estructural de preguntas y puntuación final textoCorregido = corregirPuntuacionYEstructura(textoCorregido); output.value = textoCorregido; } catch (e) { output.value = corregirPuntuacionYEstructura(texto); } } else if (estilo === 'fluido') { document.getElementById('lbl-titulo-resultado').innerText = "Mejora de Fluidez y Puntuación:"; let procesado = corregirPuntuacionYEstructura(texto); output.value = procesado; } else if (estilo === 'formal') { document.getElementById('lbl-titulo-resultado').innerText = "Texto adaptado a Estilo Formal:"; let procesado = texto; procesado = procesado.replace(/\b(hola|buenas|q tal|que tal)\b/gi, 'Estimado/a,') .replace(/\b(porfa|porfis)\b/gi, 'por favor') .replace(/\b(ok|vale|dale)\b/gi, 'de acuerdo') .replace(/\b(yo creo que|me parece que)\b/gi, 'considero que'); procesado = corregirPuntuacionYEstructura(procesado); output.value = procesado; } }
